### `harness.py`

```python
from dataclasses import dataclass
import multiprocessing as mp
import time
from typing import Callable

@dataclass
class BenchStats:
    min_ms: float
    median_ms: float
    iqr_ms: float

def fnv1a_hash(s: str) -> int:
    h = 14695981039346656037
    for b in s.encode('utf-8'):
        h = ((h ^ b) * 1099511628211) & 0xFFFFFFFFFFFFFFFF
    return h

def run_benchmark(fn: Callable[[], None], warmup: int = 2, runs: int = 7) -> BenchStats:
    for _ in range(warmup): fn()
    samples = []
    for _ in range(runs):
        t0 = time.perf_counter_ns()
        fn()
        samples.append((time.perf_counter_ns() - t0) / 1e6)
    samples.sort()
    return BenchStats(samples[0], samples[runs//2], samples[(3*runs)//4]-samples[runs//4])

def emit_verify_tag(phase: str, seed: int, m1: float, m2: float) -> str:
    hw = mp.cpu_count()
    sig = fnv1a_hash(f'{phase}|{seed}|{hw}|{m1:.4f}|{m2:.4f}')
    return f'VERIFY_HASH[{phase}]: {sig:016X} (P_log={hw})'
```

```python
import ctypes
import multiprocessing as mp
from multiprocessing import shared_memory
from harness import run_benchmark, emit_verify_tag

STUDENT_ID = 230107088
M = 16_000_000
STRIDES = (1, 2, 4, 8, 16, 32)

def fs_worker(shm_name, slot_idx, iters):
    shm = shared_memory.SharedMemory(name=shm_name)
    arr = (ctypes.c_uint64 * 512).from_buffer(shm.buf)
    acc = arr[slot_idx]
    for i in range(iters):
        acc = (acc * 6364136223846793005 + (i | 1)) & 0xFFFFFFFFFFFFFFFF
        arr[slot_idx] = acc
    del arr
    shm.close()

def trial(p, stride):
    shm = shared_memory.SharedMemory(create=True, size=4096)
    try:
        processes = []
        for k in range(p):
            count = M // p + (k < M % p)
            proc = mp.Process(target=fs_worker, args=(shm.name, k * stride, count))
            processes.append(proc)
            proc.start()
        for proc in processes:
            proc.join()
            if proc.exitcode != 0:
                raise RuntimeError(f'Worker failed: {proc.exitcode}')
    finally:
        shm.close()
        shm.unlink()

def main():
    logical = mp.cpu_count()
    physical = min(int(input(f'Enter physical CPU cores (logical={logical}): ')), logical)
    if physical < 1 or physical * max(STRIDES) > 512:
        raise ValueError('Physical core count incompatible with 4096-byte buffer')
    print('PHASE 1 - FALSE SHARING')
    sweep = {}
    for stride in STRIDES:
        stats = run_benchmark(lambda: trial(physical, stride))
        sweep[stride] = stats.median_ms
        print(f'stride={stride:2} bytes={stride*8:3} median={stats.median_ms:.2f} ms IQR={stats.iqr_ms:.2f} throughput={M/stats.median_ms/1000:.3f} Mops/s')
    counts = sorted(set([1] + [2**k for k in range(1, logical.bit_length()) if 2**k <= logical] + [logical]))
    for stride in (1, 16):
        baseline = None
        print(f'\nScaling stride={stride}')
        for p in counts:
            if p * stride >= 512: continue
            stats = run_benchmark(lambda: trial(p, stride))
            if baseline is None: baseline = stats.median_ms
            speedup = baseline / stats.median_ms
            efficiency = speedup / p
            e = ((1/speedup)-(1/p))/(1-(1/p)) if p > 1 else 0
            print(f'p={p:2} T={stats.median_ms:.2f}ms S={speedup:.3f} E={efficiency:.3f} e={e:.4f}')
    print(emit_verify_tag('PHASE1', STUDENT_ID, sweep[1], sweep[16]))

if __name__ == '__main__':
    mp.freeze_support()
    main()
```

### `task2_scaling.py`

```python
import math
import multiprocessing as mp
import time
from harness import emit_verify_tag

STUDENT_ID = 230107088
STRONG_N = 1_200_000
WEAK_N0 = 200_000

def stage_a(n):
    total = 0.0
    for i in range(max(1, n // 20)):
        x = (i + 1) * 0.00001
        for _ in range(10):
            x = math.sin(x) + math.sqrt(abs(x) + 1)
        total += x
    return total

def stage_b(args):
    start, end = args
    total = 0.0
    for i in range(start, end):
        x = (i % 1000 + 1) * 0.001
        for _ in range(30):
            x = math.sin(x) * 0.5 + math.sqrt(abs(x) + 1) * 0.5
        total += x
    return total

def trial(pool, n, p):
    t0 = time.perf_counter()
    stage_a(n)
    ta = time.perf_counter() - t0
    chunks = [(i*n//p, (i+1)*n//p) for i in range(p)]
    sum(pool.map(stage_b, chunks))
    return time.perf_counter()-t0, ta

def main():
    logical = mp.cpu_count()
    physical = int(input(f'Physical cores (logical={logical}): '))
    strong = {}
    weak = {}
    print('PHASE 2 - STRONG AND WEAK SCALING')
    for p in range(1, logical + 1):
        with mp.Pool(p) as pool:
            trial(pool, 1000, p)  # warmup
            strong[p] = trial(pool, STRONG_N, p)
            weak[p] = trial(pool, p*WEAK_N0, p)
        ts, ta = strong[p]
        tw, taw = weak[p]
        ss = strong[1][0] / ts
        ew = weak[1][0] / tw
        gust = p - (taw/tw)*(p-1)
        print(f'p={p:2} strong={ts:.3f}s S={ss:.3f} E={ss/p:.3f} weak={tw:.3f}s Eweak={ew:.3f} Gustafson={gust:.3f}')
    s = strong[1][1] / strong[1][0]
    print(f'Amdahl upper bound: {1/s:.3f}')
    if physical < logical:
        before = strong[1][0]/strong[physical][0]
        after = strong[1][0]/strong[physical+1][0]
        print(f'Marginal speedup p={physical} to {physical+1}: {after-before:.4f}')
    p = logical
    print(emit_verify_tag('PHASE2', STUDENT_ID, strong[1][0]/strong[p][0], p-(weak[p][1]/weak[p][0])*(p-1)))

if __name__ == '__main__':
```

### `task4_scheduling.py`

```python
import math
import multiprocessing as mp
import time
from harness import emit_verify_tag

STUDENT_ID = 230107088
N = 8000

def work(i):
    k = 1 + (i//40)**2
    x = 0.1
    for _ in range(k):
        x = math.sin(x) + math.cos(x)*0.1 + math.sqrt(abs(x)+1)*0.01
    return x

def worker(worker_id, p, strategy, chunk, counter, runtimes):
    start = time.perf_counter()
    if strategy == 'block':
        indices = range(worker_id*N//p, (worker_id+1)*N//p)
        for i in indices: work(i)
    elif strategy == 'cyclic':
        for i in range(worker_id, N, p): work(i)
    else:
        while True:
            with counter.get_lock():
                first = counter.value
                counter.value += chunk
            if first >= N: break
            for i in range(first, min(first+chunk, N)): work(i)
    runtimes[worker_id] = time.perf_counter()-start

def trial(p, strategy, chunk=1):
    counter = mp.Value('i', 0)
    runtimes = mp.Array('d', p)
    processes = [mp.Process(target=worker, args=(i,p,strategy,chunk,counter,runtimes)) for i in range(p)]
    t0 = time.perf_counter()
    for proc in processes: proc.start()
    for proc in processes:
        proc.join()
        if proc.exitcode: raise RuntimeError(f'Worker exit code {proc.exitcode}')
    wall = time.perf_counter()-t0
    active = list(runtimes)
    imbalance = max(active)/(sum(active)/p)
    return wall, imbalance

def main():
    physical = int(input(f'Physical CPU cores (logical={mp.cpu_count()}): '))
    print('PHASE 4 - SCHEDULING')
    results = {}
    for name, chunk in [('block',0),('cyclic',0),('dynamic',1),('dynamic',8),('dynamic',64),('dynamic',512)]:
        wall, imbalance = trial(physical,name,chunk)
        results[(name,chunk)] = wall
        print(f'{name:8} chunk={chunk:3} wall={wall:.3f}s lambda={imbalance:.3f}')
    best = min((c for c in (1,8,64,512)), key=lambda c: results[('dynamic',c)])
    print('Best dynamic chunk:',best)
    print(emit_verify_tag('PHASE4',STUDENT_ID,results[('block',0)],results[('dynamic',best)]))

if __name__ == '__main__':
    mp.freeze_support()
    main()
```
